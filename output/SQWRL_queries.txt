# Q01 — PREDIKSI SEKTOR 1-3 TAHUN (urut skor outlook skenario aktif)
Sektor(?s) ^ skorOutlookAktif(?s, ?o) -> sqwrl:select(?s, ?o) ^ sqwrl:orderByDescending(?o)

# Q02 — Sektor bullish saja
SektorBullish(?s) ^ skorOutlookAktif(?s, ?o) -> sqwrl:select(?s, ?o) ^ sqwrl:orderByDescending(?o)

# Q03 — TOP-3 EMITEN di sektor tertentu (ulangi untuk Sektor_Energi, Sektor_Kesehatan, ...)
termasukSektor(?e, Sektor_BarangBaku) ^ skorTotalEmiten(?e, ?t) -> sqwrl:select(?e, ?t) ^ sqwrl:orderByDescending(?t) ^ sqwrl:limit(3)
termasukSektor(?e, Sektor_Energi) ^ skorTotalEmiten(?e, ?t) -> sqwrl:select(?e, ?t) ^ sqwrl:orderByDescending(?t) ^ sqwrl:limit(3)
termasukSektor(?e, Sektor_Kesehatan) ^ skorTotalEmiten(?e, ?t) -> sqwrl:select(?e, ?t) ^ sqwrl:orderByDescending(?t) ^ sqwrl:limit(3)

# Q04 — Semua kandidat top pick + sektornya

# Q06 — JALUR PENJELASAN: tailwind & headwind sebuah sektor

# Q07 — Rantai kausal multi-hop dari peristiwa aktif

# Q08 — Routing geopolitik ke emiten (eksposur komoditas)

# Q10 — CASE-BASED REASONING: kemiripan periode historis

# Q11 — RAWAN COLLAPSE / going concern

# Q12 — SINYAL QUANT TRADE dengan stop loss & take profit
